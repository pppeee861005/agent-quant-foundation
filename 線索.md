import futu as ft

# 1. 定義一個「處理推送」的回呼類別
class MyQuoteHandler(ft.StockQuoteHandlerBase):
    def on_recv_rsp(self, rsp_pb):
        ret_code, data = super().on_recv_rsp(rsp_pb)
        if ret_code == ft.RET_OK:
            print(data)  # 每次有新報價就會被呼叫
        return ret_code, data

# 2. 建立連線
quote_ctx = ft.OpenQuoteContext(host="127.0.0.1", port=11111)

# 3. 註冊回呼
quote_ctx.set_handler(MyQuoteHandler())

# 4. 訂閱（這一步告訴 OpenD：請開始推送）
quote_ctx.subscribe(['HK.00700'], [ft.SubType.QUOTE])

# 5. 主程式不能結束，否則回呼也會死
import time
time.sleep(60)  # 接收 60 秒的推送

# 6. 結束前清理
quote_ctx.close()
